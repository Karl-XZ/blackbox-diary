cmake_minimum_required(VERSION 3.22.1)
project(blackbox_spark_sme2)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3")
set(VENDOR_LIBS "${CMAKE_CURRENT_LIST_DIR}/../../app/src/main/jniLibs/arm64-v8a")
foreach(name llama ggml ggml-base)
    add_library(${name} SHARED IMPORTED)
    set_target_properties(${name} PROPERTIES IMPORTED_LOCATION "${VENDOR_LIBS}/lib${name}.so")
endforeach()
add_library(blackbox_spark_runtime SHARED spark_runtime.cpp)
target_include_directories(blackbox_spark_runtime PRIVATE include)
target_link_libraries(blackbox_spark_runtime llama ggml ggml-base log dl)
set_target_properties(blackbox_spark_runtime PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${VENDOR_LIBS}")
